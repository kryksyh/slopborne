cmake_minimum_required(VERSION 3.24)
# The original guest instructions require an x86-64 host process, also on M-series Macs.
set(CMAKE_OSX_ARCHITECTURES x86_64 CACHE STRING "Guest CPU architecture")
project(bbport_macos_runtime LANGUAGES C)
if(NOT APPLE OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "x86_64")
    message(FATAL_ERROR "The macOS runtime requires an x86_64 build (Rosetta on Apple Silicon).")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
find_package(SDL3 CONFIG REQUIRED)
file(GLOB ATRAC_SOURCES CONFIGURE_DEPENDS "${ROOT}/third_party/LibAtrac9/C/src/*.c")
if(NOT ATRAC_SOURCES)
    message(FATAL_ERROR "Initialize third_party/LibAtrac9: git submodule update --init third_party/LibAtrac9")
endif()
add_library(atrac9 STATIC ${ATRAC_SOURCES})
target_compile_options(atrac9 PRIVATE -w)
file(GLOB RUNTIME_SOURCES CONFIGURE_DEPENDS "${ROOT}/src/runtime*.c")
add_library(bb-runtime STATIC ${RUNTIME_SOURCES} "${ROOT}/src/compat_macos.c")
target_include_directories(bb-runtime PUBLIC "${ROOT}" "${ROOT}/src")
target_compile_options(bb-runtime PRIVATE -Wall -Wextra -Werror -fno-omit-frame-pointer)
target_link_libraries(bb-runtime PUBLIC SDL3::SDL3 atrac9)
add_library(cpu-backend STATIC "${ROOT}/tests/cpu_backend.c")
target_include_directories(cpu-backend PRIVATE "${ROOT}")
add_executable(bb-probe-cpu "${ROOT}/src/probe.c")
target_link_libraries(bb-probe-cpu PRIVATE bb-runtime cpu-backend)
target_compile_options(bb-probe-cpu PRIVATE -Wall -Wextra -Werror -fno-omit-frame-pointer)
enable_testing()
foreach(name runtime sema macos_platform)
    add_executable(${name}-test "${ROOT}/tests/test_${name}.c")
    target_compile_options(${name}-test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    target_link_libraries(${name}-test PRIVATE bb-runtime cpu-backend)
    add_test(NAME ${name} COMMAND ${name}-test)
endforeach()
add_executable(file-mods-test "${ROOT}/tests/test_file_mods.c")
target_compile_options(file-mods-test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_test(NAME file-mods COMMAND file-mods-test)
add_executable(content-test "${ROOT}/tests/test_content.c" "${ROOT}/src/runtime_content.c")
target_include_directories(content-test PRIVATE "${ROOT}/src")
target_compile_options(content-test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_test(NAME content COMMAND content-test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME runtime-contracts
    COMMAND "${CMAKE_COMMAND}" -E env "BB_TEST_BIN_DIR=$<TARGET_FILE_DIR:bb-probe-cpu>"
    "BB_TEST_PROBE=bb-probe-cpu" "${Python3_EXECUTABLE}" -m unittest discover
    -s "${ROOT}/tests" -p "test_*runtime.py")
foreach(suite probe sema)
    add_test(NAME ${suite}-contracts
        COMMAND "${CMAKE_COMMAND}" -E env "BB_TEST_BIN_DIR=$<TARGET_FILE_DIR:bb-probe-cpu>"
        "BB_TEST_PROBE=bb-probe-cpu" "${Python3_EXECUTABLE}" -m unittest discover
        -s "${ROOT}/tests" -p "test_${suite}.py")
endforeach()

add_executable(pad-test "${ROOT}/tests/test_pad.c")
target_include_directories(pad-test PRIVATE "${ROOT}")
target_compile_options(pad-test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
target_link_libraries(pad-test PRIVATE SDL3::SDL3)
add_test(NAME pad COMMAND pad-test)
